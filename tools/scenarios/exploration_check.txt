# Exploration Peaks level check: switch from the proving ground, spawn, a road drive across a physics tile border, a
# ford, a deep-water recovery, every POI teleport, the map, both vehicles, and overview captures to compare with
# external_assets/images/Exploration.png:
#   ./build/release/trailforge --script tools/scenarios/exploration_check.txt --no-audio
# Captures /tmp/trailforge_expl_*.png. Exit code 1 on FAIL. `place X Z YAW` puts the vehicle on the ground at
# (X, Z) heading YAW degrees (0 = +Z, 90 = +X east).
set ui 0
set hud 0
set auto_reverse 0

# ---------------------------------------------------------------- level switch and spawn
expect level == 0
level exploration_peaks
until level == 1 10
wait 1.5
expect tiles_loaded >= 4
expect road_dist < 2.5
expect water_depth == 0
expect recoveries == 0
expect tilt < 12
capture /tmp/trailforge_expl_01_spawn.png
wait 0.3

# ---------------------------------------------------------------- Coast Road east across the tile border x = -255
place -300 443 88
wait 1.5
expect tile_x == 2
mark tiles_async
pedal throttle 0.5
until tile_x == 3 25
expect road_dist < 12                   # 6 m gravel road, no steering: the road curves and the car drifts down its camber
expect speed_kmh > 15
capture /tmp/trailforge_expl_02_tile_crossing.png
wait 0.3
pedal throttle 0
pedal brake 1
until speed_kmh < 1 10
pedal brake 0
expect tile_stall_ms < 30
expect tiles_loaded <= 25
expect tile_build_ms < 40
expect tile_update_ms < 30
expect d_tiles_async >= 0

# ---------------------------------------------------------------- Valley Road ford (water < 0.45 m over the road)
place -228 -333 100
wait 1.5
mark recoveries
pedal throttle 0.35
until water_depth > 0.1 15
capture /tmp/trailforge_expl_03_ford.png

capture /tmp/trailforge_expl_04_ford_side.png
wait 0.3
camera follow
until water_depth == 0 20
wait 1.0
pedal throttle 0
pedal brake 1
until speed_kmh < 1 10
pedal brake 0
expect d_recoveries == 0
expect road_dist < 5
