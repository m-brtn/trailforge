# Suspension articulation on the diagonal twister (screenshots to /tmp/trailforge_v_susp_*.png):
#   ./build/release/trailforge --script tools/scenarios/suspension.txt
# Stops on the twister and checks that the wheels articulate independently: the spread of the suspension lengths
# (whether a wheel stays in the air depends on the body: the traced bodies of v0.15-v0.22 usually settled with all
# four wheels down).
set ui 0
set auto_reverse 0               # S only brakes: the stops hold the brake at a standstill
teleport 2
wait 2.0
tap L
wait 0.3
down W
until z > -1.5 20
up W
down S
wait 2.5
telemetry stopped
expect susp_spread > 0.20
capture /tmp/trailforge_v_susp_01_follow_hud.png
wait 0.3
set hud 0
wait 0.3
camera 0.6 1.1 -5.2 -0.2 0.6 1.2
wait 0.5
capture /tmp/trailforge_v_susp_02_rear.png
wait 0.3
camera -6.8 1.0 1.2 -0.2 0.5 1.2
wait 0.5
capture /tmp/trailforge_v_susp_03_side_right.png
wait 0.3
camera 6.4 1.0 1.2 -0.2 0.5 1.2
wait 0.5
capture /tmp/trailforge_v_susp_04_side_left.png
wait 0.3
set debug_lines 1
wait 0.5
capture /tmp/trailforge_v_susp_05_side_left_debug.png
up S
quit
