# Wheel spray (cosmetic particles): calm rolling throws almost nothing, wheelspin throws a fan, a locked wheel pushes
# material ahead; per surface. Captures: /tmp/trailforge_spray_*.png. Teleports: 3 low grip (on dirt before the
# slab), 4 hills (dirt), 10 sand, 12 mud bath (faces -Z), 13 meadow, 14 puddle.
set ui 0
set hud 0
set auto_reverse 0
time Noon
quality High

# Calm rolling on packed dirt below the dust speed: almost nothing.
teleport 4
wait 1.5
set spray_clear 1
down W
wait 0.6
up W
wait 1.0
telemetry calm_roll
expect speed_kmh > 6
expect spray < 12

# Full-throttle launches (wheelspin), side-rear view 0.6 s after the throttle.
teleport 12
wait 1.5
set spray_clear 1
down W
wait 1.0
camera -37.0 1.4 47.5 -41.5 0.6 42.0
capture /tmp/trailforge_spray_mud.png
telemetry mud_spin
expect spray > 200
up W
wait 0.2

teleport 3
wait 1.5
set spray_clear 1
down W
wait 1.0
camera 4.5 1.4 15.5 0.0 0.6 21.0
capture /tmp/trailforge_spray_dirt.png
telemetry dirt_spin
expect spray > 40
up W
wait 0.2

teleport 10
wait 1.5
set spray_clear 1
down W
wait 1.0
camera 40.5 1.6 48.5 36.0 0.9 54.5
capture /tmp/trailforge_spray_sand.png
telemetry sand_spin
expect spray > 150
up W
wait 0.2

teleport 13
wait 1.5
set spray_clear 1
down W
wait 1.0
camera 23.0 2.6 45.0 19.0 0.6 50.0
capture /tmp/trailforge_spray_grass.png
telemetry grass_spin
expect spray > 50
up W
wait 0.2

# Into the first path puddle at ~20 km/h: splash.
teleport 14
wait 1.5
set spray_clear 1
down W
until z > 54.5 6
camera 9.5 2.0 51.5 4.9 0.4 56.0
capture /tmp/trailforge_spray_puddle.png
telemetry puddle
expect spray > 40
up W
wait 0.2

# Locked wheels on packed dirt: material is pushed forward.
teleport 4
wait 1.5
set spray_clear 1
down W
until speed_kmh > 25 6
up W
down S
wait 0.35
camera -21.0 2.2 -40.5 -26.0 0.6 -35.5
capture /tmp/trailforge_spray_locked.png
telemetry locked
expect spray > 5
up S
wait 0.5

# Cost at the cap: synthetic fill around the vehicle.
set spray_clear 1
set spray_fill 1
wait 0.3
expect spray >= 3000
camera follow
quit
