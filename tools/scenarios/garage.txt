# Garage: open it from the pause and the start menu, change live and rebuild values with keys, mouse and direct
# commands, save / rename / save as new / select / delete setups, back out with Escape.
#   ./build/release/trailforge --script tools/scenarios/garage.txt --no-audio
# Setups go to a scratch directory (<tmp>/trailforge_scenario_setups, emptied at start). Captures:
# /tmp/trailforge_garage_*.png

# ---------------------------------------------------------------- from the pause menu (mouse click on GARAGE)
wait 1.5
expect garage_open == 0
tap Escape
wait 0.3
expect menu_screen == 3
click menu:garage
wait 0.5
expect garage_open == 1
expect menu_screen == 4
expect paused == 0             # the simulation runs in the garage
expect cam_free == 1
expect garage_factory == 1
expect garage_setups == 0
wait 1.0
capture /tmp/trailforge_garage_01_open.png
wait 0.1

# ---------------------------------------------------------------- drivetrain: RWD (rebuild in place), then reset
garage row drivetrain.layout
wait 0.5
capture /tmp/trailforge_garage_02_drivetrain.png
wait 0.1
garage set drivetrain.layout 0
wait 0.3
expect layout == 0
expect front_driven == 0
expect rear_driven == 1
garage reset
wait 0.3
expect garage_dirty == 0
expect garage_pending == 0

# ---------------------------------------------------------------- keyboard: front spring one step up (live)
garage row suspension.front.frequency
tap Right
wait 0.2
expect garage:suspension.front.frequency > 1.39
expect garage:suspension.front.frequency < 1.41
expect cfg:suspension.front.frequency > 1.39     # reached the running vehicle
expect garage_dirty == 1
tap Down                       # rear spring
tap Left
tap Left
wait 0.2
expect cfg:suspension.rear.frequency > 1.24
expect cfg:suspension.rear.frequency < 1.26

# ---------------------------------------------------------------- mouse: click the anti-roll bar slider at 60 %
click garage:suspension.front.antiRoll:0.6
wait 0.3
expect cfg:suspension.front.antiRoll > 1000
log antiroll

# ---------------------------------------------------------------- rebuild in place: ride height +60 mm
mark y
mark x
mark z
garage set suspension.rideHeight 0.06
wait 2.0
expect cfg:suspension.rideHeight > 0.059
expect d_y > 0.04
expect d_x < 0.05
expect d_x > -0.05
expect d_z < 0.05
expect d_z > -0.05
capture /tmp/trailforge_garage_03_lifted.png
wait 0.1

# ---------------------------------------------------------------- save (mouse), rename (typed), save as new
click garage:save
wait 0.2
expect garage_setups == 1
expect garage_factory == 0
expect garage_dirty == 0
click garage:rename
wait 0.2
expect garage_renaming == 1
expect text_input == 1
tap Backspace
tap Backspace
tap Backspace
tap Backspace
tap Backspace
tap Backspace
tap Backspace
wait 0.1
text Lifted
wait 0.1
capture /tmp/trailforge_garage_04_rename.png
wait 0.1
tap Return
wait 0.2
expect garage_renaming == 0
expect text_input == 0
expect garage_open == 1
garage step tires.gripScale -1
garage save_new
expect garage_setups == 2

# ---------------------------------------------------------------- select: factory lowers the body again
mark y
garage select factory
wait 2.0
expect garage_factory == 1
expect cfg:suspension.rideHeight == 0
expect d_y < -0.04
garage select Lifted
wait 2.0
expect cfg:suspension.rideHeight > 0.059
expect cfg:tires.gripScale == 1
garage select Setup 1
wait 0.2
expect cfg:tires.gripScale < 1
garage select factory
garage step engine.torqueScale 1
expect garage_dirty == 1
garage reset
expect garage_dirty == 0

# ---------------------------------------------------------------- delete and back to the pause menu
garage select Lifted
garage delete
expect garage_setups == 1
expect garage_factory == 1
tap Escape
wait 0.3
expect garage_open == 0
expect menu_screen == 3
expect paused == 1
tap Escape
wait 0.3
expect menu_screen == 2
expect cam_free == 0           # the follow camera is back

# ---------------------------------------------------------------- from the start menu (keyboard)
menu main
wait 0.5
menu select garage
tap Return
wait 0.5
expect garage_open == 1
garage row vehicle
wait 1.5
capture /tmp/trailforge_garage_05_start_menu.png
wait 0.1
# mouse wheel over the long parameter list: scrolls without moving the cursor, back up to the top
expect garage_scroll == 0
wheel -4
wait 0.5
log garage_wheel_down
expect garage_scroll > 100
expect garage_cursor == 0
capture /tmp/trailforge_garage_06_wheel.png
wait 0.1
wheel 20
wait 0.5
expect garage_scroll == 0
tap Down                       # a keyboard move after the wheel still works
wait 0.2
expect garage_cursor == 1
tap Escape
wait 0.3
expect menu_screen == 1
expect menu_cursor == 1
expect garage_open == 0
